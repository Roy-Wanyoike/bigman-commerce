   217→function flattenCategories(nodes: FlatCategory[], prefix = ''): { id: string; name: string }[] {
   218→  const result: { id: string; name: string }[] = []
   219→  for (const node of nodes) {
   220→    const label = prefix ? `${prefix} > ${node.name}` : node.name
   221→    result.push({ id: node.id, name: label })
   222→    if (node.children?.length) {
   223→      result.push(...flattenCategories(node.children, label))
   224→    }
   225→  }
   226→  return result
   227→}

   229→function getStockInfo(stockCount: number, lowStockThreshold: number, trackInventory: boolean) {
   230→  if (!trackInventory) return { label: 'Not Tracked', color: 'text-muted-foreground', bg: 'bg-gray-50' }
   231→  if (stockCount <= 0) return { label: 'Out of Stock', color: 'text-red-700', bg: 'bg-red-50' }
   232→  if (stockCount <= lowStockThreshold) return { label: 'Low Stock', color: 'text-amber-700', bg: 'bg-amber-50' }
   233→  return { label: 'In Stock', color: 'text-emerald-700', bg: 'bg-emerald-50' }
   234→}

   553→  async function handleFileUpload(e: React.ChangeEvent<HTMLInputElement>) {
   554→    const files = e.target.files
   555→    if (!files || files.length === 0) return
   556→
   557→    const imageCount = product?.productImages?.length || 0
   558→    const remaining = 5 - imageCount
   559→    const filesToUpload = Array.from(files).slice(0, remaining)
   560→
   561→    if (files.length > remaining) {
   562→      alert(`Only ${remaining} more image(s) can be uploaded. ${files.length - remaining} file(s) skipped.`)
   563→    }
   564→
   565→    setUploading(true)
   566→    for (const file of filesToUpload) {
   567→      const formData = new FormData()
   568→      formData.append('file', file)
   569→      formData.append('productId', id)
   570→      formData.append('imageType', 'FRONT')
   571→      formData.append('altText', product?.name || '')
   572→      formData.append('isPrimary', String(imageCount === 0))
   573→
   574→      try {
   575→        const res = await fetch('/api/admin/images/upload', {
   576→          method: 'POST',
   577→          body: formData,
   578→        })
   579→        if (!res.ok) {
   580→          const err = await res.json()
   581→          alert(`Upload failed for ${file.name}: ${err.error}`)
   582→          break
   583→        }
   584→      } catch {
   585→        alert(`Upload failed for ${file.name}`)
   586→        break
   587→      }
   588→    }
   589→    setUploading(false)
   590→
   591→    // Reset file input
   592→    if (fileInputRef.current) fileInputRef.current.value = ''
   593→
   594→    // Reload product
   595→    const prodRes = await fetch(`/api/admin/products/${id}`).then((r) => r.json())
   596→    if (prodRes.product) setProduct(prodRes.product)
   597→  }

   614→  function formatFileSize(bytes: number | null) {
   615→    if (!bytes) return '—'
