     8→// ── helpers ──────────────────────────────────────────────────────────────────
     9→function slug(name: string): string {
    10→  return name
    11→    .toLowerCase()
    12→    .replace(/[^a-z0-9]+/g, '-')
    13→    .replace(/-+/g, '-')
    14→    .replace(/^-|-$/g, '')
    15→}
    16→
    17→function toKebab(name: string): string {
    18→  return slug(name)
    19→}
    20→
    21→function compareAt(base: number, pct: number): number {
    22→  return Math.round(base * (1 + pct / 100))
    23→}
    24→
    25→function rand(min: number, max: number): number {
    26→  return Math.floor(Math.random() * (max - min + 1)) + min
    27→}

   593→async function main() {
   594→  console.log('🔍 Looking up brands and categories...')
   595→
   596→  // Build brand lookup
   597→  const allBrands = await db.brand.findMany({ select: { id: true, name: true } })
   598→  const brandMap = new Map(allBrands.map(b => [b.name, b.id]))
   599→
   600→  // Build category lookup (both simple slug and parent/slug composite)
   601→  const allCats = await db.category.findMany({ select: { id: true, slug: true, parentId: true } })
   602→  const catMap = new Map<string, string>()
   603→  for (const c of allCats) {
   604→    catMap.set(c.slug, c.id)
   605→    const parent = c.parentId ? allCats.find(p => p.id === c.parentId) : null
   606→    if (parent) {
   607→      catMap.set(`${parent.slug}/${c.slug}`, c.id)
   608→    }
   609→  }
   610→
   611→  console.log(`   Brands: ${brandMap.size} | Categories: ${allCats.length}`)
   612→
   613→  // Step 1: Update any DRAFT products to ACTIVE
   614→  console.log('\n📦 Updating any DRAFT products to ACTIVE...')
   615→  const draftUpdate = await db.product.updateMany({
   616→    where: { status: 'DRAFT' },
   617→    data: { status: 'ACTIVE', publishedAt: new Date() },
   618→  })
   619→  console.log(`   Updated ${draftUpdate.count} DRAFT products → ACTIVE`)
   620→
   621→  // Step 2: Get business unit ID
   622→  const bu = await db.businessUnit.findFirst({ where: { type: 'COMPUTERS' } })
   623→  const businessUnitId = bu?.id || null
   624→
   625→  // Step 3: Seed products
   626→  console.log(`\n🚀 Seeding ${products.length} products...`)
   627→
   628→  let created = 0
   629→  let skipped = 0
   630→  let imagesCreated = 0
   631→
   632→  for (let i = 0; i < products.length; i++) {
   633→    const p = products[i]
   634→    const productSlug = slug(p.name)
   635→
   636→    // Check if already exists
   637→    const existing = await db.product.findUnique({ where: { slug: productSlug } })
   638→    if (existing) {
   639→      skipped++
   640→      console.log(`   ⏭️  [${i + 1}/${products.length}] SKIP (exists): ${p.name}`)
   641→      continue
   642→    }
   643→
   644→    // Resolve brand ID
   645→    const brandId = brandMap.get(p.brand) || null
   646→
   647→    // Resolve category IDs
   648→    const categoryConnects: { categoryId: string; sortOrder: number }[] = []
   649→    for (const catSlug of p.categories) {
   650→      const catId = catMap.get(catSlug)
   651→      if (catId) {
   652→        categoryConnects.push({ categoryId: catId, sortOrder: categoryConnects.length })
   653→      }
   654→    }
   655→
   656→    if (categoryConnects.length === 0) {
   657→      console.log(`   ⚠️  [${i + 1}/${products.length}] NO CATEGORIES FOUND: ${p.name} (${p.categories.join(', ')})`)
   658→      skipped++
   659→      continue
   660→    }
   661→
   662→    const compareAtPrice = compareAt(p.basePrice, p.compareAtPct)
   663→
   664→    // Create product with images

   668→        slug: productSlug,

   675→        basePrice: p.basePrice,

   677→        currency: 'KES',
   678→        warrantyMonths: p.warrantyMonths,
   679→        specifications: JSON.stringify(p.specifications),
   680→        stockCount: p.stockCount,
   681→        trackInventory: true,
   682→        lowStockThreshold: 5,
   683→        status: 'ACTIVE',
   684→        publishedAt: new Date(),
   685→        isFeatured: p.isFeatured,
   686→        isDeal: p.isDeal,
   687→        isGaming: p.isGaming,
   688→        dealLabel: p.dealLabel || null,
   689→        seoTitle: p.seoTitle,
   690→        seoDescription: p.seoDescription,
   691→        sku: p.sku,
   692→        categories: {
   693→          create: categoryConnects,
   694→        },

   711→            }
   712→          }),
   713→        },
   714→      },
