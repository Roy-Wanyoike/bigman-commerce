    58→    58→/** Walk the category tree to find a node matching the slug path */
    59→    59→function findCategoryByPath(tree: CategoryNode[], slugs: string[]): CategoryNode | null {
    60→    60→  if (!slugs.length) return null
    61→    61→  let current = tree.find(c => c.slug === slugs[0]) || null
    62→    62→  for (let i = 1; i < slugs.length; i++) {
    63→    63→    if (!current) return null
    64→    64→    current = current.children.find(c => c.slug === slugs[i]) || null
    65→    65→  }
    66→    66→  return current
    67→    67→}
    68→    68→
    69→    69→/** Build breadcrumb trail from slug array + category tree */
    70→    70→function buildBreadcrumbs(tree: CategoryNode[], slugs: string[]) {
    71→    71→  const crumbs: { name: string; href: string }[] = [{ name: 'Home', href: '/' }]
    72→    72→  let nodes: CategoryNode[] = tree
    73→    73→  for (let i = 0; i < slugs.length; i++) {
    74→    74→    const found = nodes.find(c => c.slug === slugs[i])
    75→    75→    const href = '/shop/' + slugs.slice(0, i + 1).join('/')
    76→    76→    crumbs.push({ name: found?.name || slugs[i], href })
    77→    77→    if (found) nodes = found.children
    78→    78→  }
    79→    79→  return crumbs
    80→    80→}
    81→    81→
    82→    82→/** Build products API URL from filter state */
    83→    83→function buildProductsUrl(opts: {
    84→    84→  categorySlug: string | null
    85→    85→  brands: string[]
    86→    86→  conditions: string[]
    87→    87→  minPrice: string
    88→    88→  maxPrice: string
    89→    89→  sort: string
    90→    90→  page: number
    91→    91→}) {
    92→    92→  const params = new URLSearchParams()
    93→    93→  if (opts.categorySlug) params.set('category', opts.categorySlug)
    94→    94→  if (opts.brands.length > 0) params.set('brand', opts.brands.join(','))
    95→    95→  if (opts.conditions.length > 0) params.set('condition', opts.conditions.join(','))
    96→    96→  if (opts.minPrice) params.set('minPrice', opts.minPrice)
    97→    97→  if (opts.maxPrice) params.set('maxPrice', opts.maxPrice)
    98→    98→  if (opts.sort) params.set('sort', opts.sort)
    99→    99→  params.set('limit', String(PAGE_SIZE))
   100→   100→  params.set('offset', String((opts.page - 1) * PAGE_SIZE))
   101→   101→  return `/api/products?${params.toString()}`
   102→   102→}
