   190→// Flat list of all categories for multi-select

   200→function flattenCategories(nodes: FlatCategory[], prefix = ''): { id: string; name: string }[] {
   201→  const result: { id: string; name: string }[] = []
   202→  for (const node of nodes) {
   203→    const label = prefix ? `${prefix} > ${node.name}` : node.name
   204→    result.push({ id: node.id, name: label })
   205→    if (node.children?.length) {
   206→      result.push(...flattenCategories(node.children, label))
   207→    }
   208→  }
   209→  return result
   210→}
   211→
   212→function getStockInfo(stockCount: number, lowStockThreshold: number, trackInventory: boolean) {
   213→  if (!trackInventory) return { label: 'Not Tracked', color: 'text-muted-foreground', bg: 'bg-gray-50' }
   214→  if (stockCount <= 0) return { label: 'Out of Stock', color: 'text-red-700', bg: 'bg-red-50' }
   215→  if (stockCount <= lowStockThreshold) return { label: 'Low Stock', color: 'text-amber-700', bg: 'bg-amber-50' }
   216→  return { label: 'In Stock', color: 'text-emerald-700', bg: 'bg-emerald-50' }
   217→}
