   194→// FULL CATALOG WITH FILTERS
   195→// ============================================================
   196→export function FullCatalog({ products, categories, brands }: { products: Product[]; categories: CategoryNode[]; brands: Brand[] }) {
   197→  const [activeCat, setActiveCat] = useState<string | null>(null)
   198→  const [sortBy, setSortBy] = useState('sortOrder')
   199→  const [conditionFilter, setConditionFilter] = useState<string | null>(null)
   200→  const { compareList, clearCompare } = useStore()
   201→
   202→  let filtered = [...products]
   203→  if (activeCat) {
   204→    const catSlugs = getCatAndDescendantSlugs(categories, activeCat)
   205→    filtered = filtered.filter(p => p.categories.some(c => catSlugs.includes(c.category.slug)))
   206→  }
   207→  if (conditionFilter) filtered = filtered.filter(p => p.condition === conditionFilter)
   208→
   209→  if (sortBy === 'price-asc') filtered.sort((a, b) => a.basePrice - b.basePrice)
   210→  else if (sortBy === 'price-desc') filtered.sort((a, b) => b.basePrice - a.basePrice)
   211→  else if (sortBy === 'newest') filtered.sort((a, b) => (b.publishedAt?.getTime() || 0) - (a.publishedAt?.getTime() || 0))
   212→
   213→  const topCategories = categories.filter(c => !c.parentId)
   214→
   215→  return (
   216→    <section className="py-10 md:py-14 bg-muted/30">
   217→      <div className="container-main">
   218→        <div className="flex items-end justify-between mb-6">
   219→          <div>
   220→            <h2 className="text-xl md:text-2xl font-bold tracking-tight">All Products</h2>
   221→            <p className="text-sm text-muted-foreground mt-0.5">{filtered.length} products available</p>
   222→          </div>
   223→        </div>
   224→
   225→        {/* Filters bar */}
   226→        <div className="min-w-0 flex flex-wrap items-center gap-2 mb-6 p-3 bg-card rounded-xl border border-border/60">
   227→          <div className="overflow-x-auto flex-nowrap flex items-center gap-2 min-w-0">
   228→          <SlidersHorizontal className="h-4 w-4 text-muted-foreground mr-1 shrink-0" />
   229→          <Button variant={activeCat === null ? 'default' : 'outline'} size="sm" className="h-7 text-xs rounded-md shrink-0" onClick={() => setActiveCat(null)}>All</Button>
   230→          {topCategories.slice(0, 10).map(c => (
   231→            <Button key={c.id} variant={activeCat === c.slug ? 'default' : 'outline'} size="sm" className="h-7 text-xs rounded-md shrink-0"
   232→              onClick={() => setActiveCat(activeCat === c.slug ? null : c.slug)}>
   233→              {c.name}
   234→            </Button>
   235→          ))}
   236→          </div>
   237→          <div className="ml-auto flex items-center gap-2 shrink-0">
   238→            <Select value={conditionFilter || 'all'} onValueChange={v => setConditionFilter(v === 'all' ? null : v)}>
   239→              <SelectTrigger className="h-7 w-32 text-xs"><SelectValue placeholder="Condition" /></SelectTrigger>
   240→              <SelectContent>
   241→                <SelectItem value="all">All Conditions</SelectItem>
   242→                <SelectItem value="NEW">New</SelectItem>
   243→                <SelectItem value="REFURBISHED">Refurbished</SelectItem>
   244→              </SelectContent>
   245→            </Select>
   246→            <Select value={sortBy} onValueChange={setSortBy}>
   247→              <SelectTrigger className="h-7 w-36 text-xs"><ArrowUpDown className="h-3 w-3 mr-1" /><SelectValue /></SelectTrigger>
   248→              <SelectContent>
   249→                <SelectItem value="sortOrder">Featured</SelectItem>
   250→                <SelectItem value="price-asc">Price: Low → High</SelectItem>
   251→                <SelectItem value="price-desc">Price: High → Low</SelectItem>
   252→                <SelectItem value="newest">Newest</SelectItem>
   253→              </SelectContent>
   254→            </Select>
   255→          </div>
   256→        </div>
   257→
   258→        {/* Product grid */}
   259→        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3 md:gap-4">
   260→          {filtered.map(p => <ProductCard key={p.id} product={p} />)}
   261→        </div>
   262→
   263→        {filtered.length === 0 && (
   264→          <div className="text-center py-16 text-muted-foreground">
   265→            <Search className="h-10 w-10 mx-auto mb-3 opacity-30" />
   266→            <p className="text-base font-medium">No products found</p>
   267→            <p className="text-sm mt-1">Try adjusting your filters</p>
   268→          </div>
   269→        )}
   270→      </div>
   271→    </section>
   272→  )
   273→}
   274→
   275→function getCatAndDescendantSlugs(categories: CategoryNode[], slug: string): string[] {
   276→  const cat = categories.find(c => c.slug === slug)
   277→  if (!cat) return [slug]
   278→  const slugs = [slug]
   279→  function walk(children: CategoryNode[]) {
   280→    for (const c of children) { slugs.push(c.slug); if (c.children?.length) walk(c.children) }
   281→  }
   282→  if (cat.children?.length) walk(cat.children)
   283→  return slugs
   284→}
