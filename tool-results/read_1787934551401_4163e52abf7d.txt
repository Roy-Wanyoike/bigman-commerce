    60→/** Walk the category tree to find a node matching the slug path */
    61→function findCategoryByPath(tree: CategoryNode[], slugs: string[]): CategoryNode | null {
    62→  if (!slugs.length) return null
    63→  let current = tree.find(c => c.slug === slugs[0]) || null
    64→  for (let i = 1; i < slugs.length; i++) {
    65→    if (!current) return null
    66→    current = current.children.find(c => c.slug === slugs[i]) || null
    67→  }
    68→  return current
    69→}
    70→
    71→/** Build breadcrumb trail from slug array + category tree */
    72→function buildBreadcrumbs(tree: CategoryNode[], slugs: string[]) {
    73→  const crumbs: { name: string; href: string }[] = [{ name: 'Home', href: '/' }]
    74→  let nodes: CategoryNode[] = tree
    75→  for (let i = 0; i < slugs.length; i++) {
    76→    const found = nodes.find(c => c.slug === slugs[i])
    77→    const href = '/shop/' + slugs.slice(0, i + 1).join('/')
    78→    crumbs.push({ name: found?.name || slugs[i], href })
    79→    if (found) nodes = found.children
    80→  }
    81→  return crumbs
    82→}
    83→
    84→/** Build products API URL from filter state */
    85→function buildProductsUrl(opts: {
    86→  categorySlug: string | null
    87→  brands: string[]
    88→  conditions: string[]
    89→  minPrice: string
    90→  maxPrice: string
    91→  sort: string
    92→  page: number
    93→}) {
    94→  const params = new URLSearchParams()
    95→  if (opts.categorySlug) params.set('category', opts.categorySlug)
    96→  if (opts.brands.length > 0) params.set('brand', opts.brands.join(','))
    97→  if (opts.conditions.length > 0) params.set('condition', opts.conditions.join(','))
    98→  if (opts.minPrice) params.set('minPrice', opts.minPrice)
    99→  if (opts.maxPrice) params.set('maxPrice', opts.maxPrice)
   100→  if (opts.sort) params.set('sort', opts.sort)
   101→  params.set('limit', String(PAGE_SIZE))
   102→  params.set('offset', String((opts.page - 1) * PAGE_SIZE))
   103→  return `/api/products?${params.toString()}`
   104→}

   499→                {/* Pagination */}
   500→                {totalPages > 1 && (
   501→                  <nav className="flex items-center justify-center gap-1.5 mt-8 mb-4" aria-label="Pagination">
   502→                    <Button
   503→                      variant="outline" size="icon" className="h-9 w-9"
   504→                      disabled={page <= 1}
   505→                      onClick={() => setPage(p => Math.max(1, p - 1))}
   506→                    >
   507→                      <ChevronLeft className="h-4 w-4" />
   508→                    </Button>
   509→
   510→                    {pageNumbers.map((p, i) =>
   511→                      p === -1 ? (
   512→                        <span key={`ellipsis-${i}`} className="px-1.5 text-muted-foreground">…</span>
   513→                      ) : (
   514→                        <Button
   515→                          key={p}
   516→                          variant={p === page ? 'default' : 'outline'}
   517→                          size="icon"
   518→                          className="h-9 w-9"
   519→                          onClick={() => setPage(p)}
   520→                        >
   521→                          {p}
   522→                        </Button>
   523→                      )
   524→                    )}
   525→
   526→                    <Button
   527→                      variant="outline" size="icon" className="h-9 w-9"
   528→                      disabled={page >= totalPages}
   529→                      onClick={() => setPage(p => Math.min(totalPages, p + 1))}
   530→                    >
   531→                      <ChevronR className="h-4 w-4" />
   532→                    </Button>
   533→                  </nav>
   534→                )}
   535→              </>
   536→            )}
   537→          </div>
   538→        </div>
   539→      </div>
   540→      </main>
   541→      <BigmanFooter categories={categories} />
   542→      <MobileBottomNav />
   543→      <div className="lg:hidden h-14" />
   544→    </div>
   545→  )
   546→}
