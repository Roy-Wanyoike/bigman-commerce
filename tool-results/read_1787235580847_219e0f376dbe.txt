   194→// FULL CATALOG WITH FILTERS
   195→// ============================================================
   196→export function FullCatalog({ products, categories, brands }: { products: Product[]; categories: CategoryNode[]; brands: Brand[] }) {
   197→  const [activeCat, setActiveCat] = useState<string | null>(null)
   198→  const [sortBy, setSortBy] = useState('sortOrder')
   199→  const [conditionFilter, setConditionFilter] = useState<string | null>(null)
   200→  const { compareList, clearCompare } = useStore()
   201→
   202→  let filtered = [...products]
   203→  if (activeCat) {
   204→    const catSlugs = getCatAndDescendantSlugs(categories, activeCat)
   205→    filtered = filtered.filter(p => p.categories.some(c => catSlugs.includes(c.category.slug)))
   206→  }
   207→  if (conditionFilter) filtered = filtered.filter(p => p.condition === conditionFilter)
   208→
   209→  if (sortBy === 'price-asc') filtered.sort((a, b) => a.basePrice - b.basePrice)
   210→  else if (sortBy === 'price-desc') filtered.sort((a, b) => b.basePrice - a.basePrice)
   211→  else if (sortBy === 'newest') filtered.sort((a, b) => (b.publishedAt?.getTime() || 0) - (a.publishedAt?.getTime() || 0))
   212→
   213→  const topCategories = categories.filter(c => !c.parentId)
   214→
   215→  return (
   216→    <section className="py-10 md:py-14 bg-muted/30">
   217→      <div className="max-w-7xl mx-auto px-4">
   218→        <div className="flex items-end justify-between mb-6">
   219→          <div>
   220→            <h2 className="text-xl md:text-2xl font-bold tracking-tight">All Products</h2>
   221→            <p className="text-sm text-muted-foreground mt-0.5">{filtered.length} products available</p>
   222→          </div>
   223→        </div>
   224→
   225→        {/* Filters bar */}
   226→        <div className="flex flex-wrap items-center gap-2 mb-6 p-3 bg-card rounded-xl border border-border/60">
   227→          <SlidersHorizontal className="h-4 w-4 text-muted-foreground mr-1" />
   228→          <Button variant={activeCat === null ? 'default' : 'outline'} size="sm" className="h-7 text-xs rounded-md" onClick={() => setActiveCat(null)}>All</Button>
   229→          {topCategories.slice(0, 10).map(c => (
   230→            <Button key={c.id} variant={activeCat === c.slug ? 'default' : 'outline'} size="sm" className="h-7 text-xs rounded-md"
   231→              onClick={() => setActiveCat(activeCat === c.slug ? null : c.slug)}>
   232→              {c.name}
   233→            </Button>
   234→          ))}
   235→          <div className="ml-auto flex items-center gap-2">
   236→            <Select value={conditionFilter || 'all'} onValueChange={v => setConditionFilter(v === 'all' ? null : v)}>
   237→              <SelectTrigger className="h-7 w-32 text-xs"><SelectValue placeholder="Condition" /></SelectTrigger>
   238→              <SelectContent>
   239→                <SelectItem value="all">All Conditions</SelectItem>
   240→                <SelectItem value="NEW">New</SelectItem>
   241→                <SelectItem value="REFURBISHED">Refurbished</SelectItem>
   242→              </SelectContent>
   243→            </Select>
   244→            <Select value={sortBy} onValueChange={setSortBy}>
   245→              <SelectTrigger className="h-7 w-36 text-xs"><ArrowUpDown className="h-3 w-3 mr-1" /><SelectValue /></SelectTrigger>
   246→              <SelectContent>
   247→                <SelectItem value="sortOrder">Featured</SelectItem>
   248→                <SelectItem value="price-asc">Price: Low → High</SelectItem>
   249→                <SelectItem value="price-desc">Price: High → Low</SelectItem>
   250→                <SelectItem value="newest">Newest</SelectItem>
   251→              </SelectContent>
   252→            </Select>
   253→          </div>
   254→        </div>
   255→
   256→        {/* Product grid */}
   257→        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3 md:gap-4">
   258→          {filtered.map(p => <ProductCard key={p.id} product={p} />)}
   259→        </div>
   260→
   261→        {filtered.length === 0 && (
   262→          <div className="text-center py-16 text-muted-foreground">
   263→            <Search className="h-10 w-10 mx-auto mb-3 opacity-30" />
   264→            <p className="text-base font-medium">No products found</p>
   265→            <p className="text-sm mt-1">Try adjusting your filters</p>
   266→          </div>
   267→        )}
   268→      </div>
   269→    </section>
   270→  )
   271→}
   272→
   273→function getCatAndDescendantSlugs(categories: CategoryNode[], slug: string): string[] {
   274→  const cat = categories.find(c => c.slug === slug)
   275→  if (!cat) return [slug]
   276→  const slugs = [slug]
   277→  function walk(children: CategoryNode[]) {
   278→    for (const c of children) { slugs.push(c.slug); if (c.children?.length) walk(c.children) }
   279→  }
   280→  if (cat.children?.length) walk(cat.children)
   281→  return slugs
   282→}
