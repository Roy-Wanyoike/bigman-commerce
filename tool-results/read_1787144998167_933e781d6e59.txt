   238→  async function insertCategories(cats: CatDef[], parentId: string | null, level: number) {
   239→    for (let i = 0; i < cats.length; i++) {
   240→      const c = cats[i]
   241→      const cat = await db.category.create({
   242→        data: {
   243→          name: c.name, slug: c.slug,
   244→          description: c.desc || `${c.name} at Bigman Computers - Quality technology at competitive prices.`,
   245→          parentId, sortOrder: i, level, isActive: true,
   246→          isFeatured: c.isFeatured || false,
   247→          navIcon: c.icon || null, navColumns: c.navCols || 3,
   248→          showInNav: true,
   249→          seoTitle: `${c.name} in Kenya | Bigman Computers`,
   250→          seoDescription: `Shop ${c.name.toLowerCase()} at Bigman Computers, Nairobi. ${c.desc || `Quality ${c.name.toLowerCase()} at competitive prices.`}`,
   251→        }
   252→      })
   253→      catIds[c.slug] = cat.id
   254→      if (c.children?.length) {
   255→        await insertCategories(c.children, cat.id, level + 1)
   256→      }
   257→    }
   258→  }
   259→
   260→  await insertCategories(categories, null, 0)
