    58→/** Walk the category tree to find a node matching the slug path */
    59→function findCategoryByPath(tree: CategoryNode[], slugs: string[]): CategoryNode | null {
    60→  if (!slugs.length) return null
    61→  let current = tree.find(c => c.slug === slugs[0]) || null
    62→  for (let i = 1; i < slugs.length; i++) {
    63→    if (!current) return null
    64→    current = current.children.find(c => c.slug === slugs[i]) || null
    65→  }
    66→  return current
    67→}
    68→
    69→/** Build breadcrumb trail from slug array + category tree */
    70→function buildBreadcrumbs(tree: CategoryNode[], slugs: string[]) {
    71→  const crumbs: { name: string; href: string }[] = [{ name: 'Home', href: '/' }]
    72→  let nodes: CategoryNode[] = tree
    73→  for (let i = 0; i < slugs.length; i++) {
    74→    const found = nodes.find(c => c.slug === slugs[i])
    75→    const href = '/shop/' + slugs.slice(0, i + 1).join('/')
    76→    crumbs.push({ name: found?.name || slugs[i], href })
    77→    if (found) nodes = found.children
    78→  }
    79→  return crumbs
    80→}

    82→/** Build products API URL from filter state */
    83→function buildProductsUrl(opts: {
    84→  categorySlug: string | null
    85→  brands: string[]
    86→  conditions: string[]
    87→  minPrice: string
    88→  maxPrice: string
    89→  sort: string
    90→  page: number
    91→}) {
    92→  const params = new URLSearchParams()
    93→  if (opts.categorySlug) params.set('category', opts.categorySlug)
    94→  if (opts.brands.length > 0) params.set('brand', opts.brands.join(','))
    95→  if (opts.conditions.length > 0) params.set('condition', opts.conditions.join(','))
    96→  if (opts.minPrice) params.set('minPrice', opts.minPrice)
    97→  if (opts.maxPrice) params.set('maxPrice', opts.maxPrice)
    98→  if (opts.sort) params.set('sort', opts.sort)
    99→  params.set('limit', String(PAGE_SIZE))
   100→  params.set('offset', String((opts.page - 1) * PAGE_SIZE))
   101→  return `/api/products?${params.toString()}`
   102→}

   500→                    {pageNumbers.map((p, i) =>
   501→                      p === -1 ? (
   502→                        <span key={`ellipsis-${i}`} className="px-1.5 text-muted-foreground">…</span>
   503→                      ) : (
   504→                        <Button
   505→                          key={p}
   506→                          variant={p === page ? 'default' : 'outline'}
   507→                          size="icon"
   508→                          className="h-9 w-9"
   509→                          onClick={() => setPage(p)}
   510→                        >
   511→                          {p}
   512→                        </Button>
   513→                      )
   514→                    )}
   515→
   516→                    <Button
   517→                      variant="outline" size="icon" className="h-9 w-9"
   518→                      disabled={page >= totalPages}
   519→                      onClick={() => setPage(p => Math.min(totalPages, p + 1))}
   520→                    >
   521→                      <ChevronR className="h-4 w-4" />
   522→                    </Button>
   523→                  </nav>
   524→                )}
   525→              </>
   526→            )}
   527→          </div>
   528→        </div>
   529→      </div>
   530→    </main>
   531→  )
   532→}
