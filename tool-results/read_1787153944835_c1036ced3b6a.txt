   204→   204→
   205→   205→function flattenCategories(nodes: FlatCategory[], prefix = ''): { id: string; name: string }[] {
   206→   206→  const result: { id: string; name: string }[] = []
   207→   207→  for (const node of nodes) {
   208→   208→    const label = prefix ? `${prefix} > ${node.name}` : node.name
   209→   209→    result.push({ id: node.id, name: label })
   210→   210→    if (node.children?.length) {
   211→   211→      result.push(...flattenCategories(node.children, label))
   212→   212→    }
   213→   213→  }
   214→   214→  return result
   215→   215→}
