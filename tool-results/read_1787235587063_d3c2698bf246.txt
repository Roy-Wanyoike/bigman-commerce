   196→   196→export function FullCatalog({ products, categories, brands }: { products: Product[]; categories: CategoryNode[]; brands: Brand[] }) {
   197→   197→  const [activeCat, setActiveCat] = useState<string | null>(null)
   198→   198→  const [sortBy, setSortBy] = useState('sortOrder')
   199→   199→  const [conditionFilter, setConditionFilter] = useState<string | null>(null)
   200→   200→  const { compareList, clearCompare } = useStore()
   201→   201→
   202→   202→  let filtered = [...products]
   203→   203→  if (activeCat) {
   204→   204→    const catSlugs = getCatAndDescendantSlugs(categories, activeCat)
   205→   205→    filtered = filtered.filter(p => p.categories.some(c => catSlugs.includes(c.category.slug)))
   206→   206→  }
   207→   207→  if (conditionFilter) filtered = filtered.filter(p => p.condition === conditionFilter)
   208→   208→

   272→   272→
   273→   273→function getCatAndDescendantSlugs(categories: CategoryNode[], slug: string): string[] {
   274→   274→  const cat = categories.find(c => c.slug === slug)
   275→   275→  if (!cat) return [slug]
   276→   276→  const slugs = [slug]
   277→   277→  function walk(children: CategoryNode[]) {
   278→   278→    for (const c of children) { slugs.push(c.slug); if (c.children?.length) walk(c.children) }
   279→   279→  }
   280→   280→  if (cat.children?.length) walk(cat.children)
   281→   281→  return slugs
   282→   282→}
