   807→                      <TableCell colSpan={5} className="text-center py-6 text-muted-foreground">No price changes recorded.</TableCell>
   808→                    </TableRow>
   809→                  ) : (
   810→                    product.priceHistory.map((h) => (
   811→                      <TableRow key={h.id}>
   812→                        <TableCell className="font-medium text-sm">{h.priceField}</TableCell>
   813→                        <TableCell className="text-sm">{h.previousPrice != null ? formatPrice(h.previousPrice) : '—'}</TableCell>
   814→                        <TableCell className="text-sm">{formatPrice(h.newPrice)}</TableCell>
   815→                        <TableCell className="text-sm text-muted-foreground">{h.reason || '—'}</TableCell>
   816→                        <TableCell className="hidden sm:table-cell text-sm text-muted-foreground">{new Date(h.createdAt).toLocaleDateString()}</TableCell>
   817→                      </TableRow>
   818→                    ))
   819→                  )}
   820→                </TableBody>
   821→              </Table>
   822→            </CardContent>
   823→          </Card>
   824→        </div>
   825→
   826→        {/* Sidebar - 1 col */}
   827→        <div className="space-y-6">
   828→          {/* Status & Flags */}
   829→          <Card>
   830→            <CardHeader><CardTitle className="text-base">Status & Flags</CardTitle></CardHeader>
   831→            <CardContent className="space-y-4">
   832→              <div className="space-y-2">
   833→                <Label>Current Status</Label>
   834→                <Badge variant="outline" className={statusColor[product.status] || ''}>
   835→                  {statusLabel[product.status] || product.status}
   836→                </Badge>
   837→              </div>
   838→              <Separator />
   839→              <div className="flex items-center justify-between">
   840→                <Label htmlFor="isFeatured">Featured</Label>
   841→                <Switch id="isFeatured" checked={form.isFeatured as boolean} onCheckedChange={(v) => updateField('isFeatured', v)} />
   842→              </div>
   843→              <div className="flex items-center justify-between">
   844→                <Label htmlFor="isDeal">Deal</Label>
   845→                <Switch id="isDeal" checked={form.isDeal as boolean} onCheckedChange={(v) => updateField('isDeal', v)} />
   846→              </div>
   847→              <div className="flex items-center justify-between">
   848→                <Label htmlFor="isGaming">Gaming</Label>
   849→                <Switch id="isGaming" checked={form.isGaming as boolean} onCheckedChange={(v) => updateField('isGaming', v)} />
   850→              </div>
   851→            </CardContent>
   852→          </Card>
   853→
   854→          {/* Inventory */}
   855→          <Card>
   856→            <CardHeader><CardTitle className="text-base">Inventory</CardTitle></CardHeader>
   857→            <CardContent className="space-y-4">
   858→              <div className="flex items-center justify-between">
   859→                <Label htmlFor="trackInventory">Track Inventory</Label>
   860→                <Switch id="trackInventory" checked={form.trackInventory as boolean} onCheckedChange={(v) => updateField('trackInventory', v)} />
   861→              </div>
   862→              {form.trackInventory && (
   863→                <>
   864→                  <div className="space-y-2">
   865→                    <Label>Stock Count</Label>
   866→                    <Input type="number" value={form.stockCount as number} onChange={(e) => updateField('stockCount', parseInt(e.target.value) || 0)} />
   867→                  </div>
   868→                  <div className="space-y-2">
   869→                    <Label>Low Stock Threshold</Label>
   870→                    <Input type="number" value={form.lowStockThreshold as number} onChange={(e) => updateField('lowStockThreshold', parseInt(e.target.value) || 5)} />
   871→                  </div>
   872→                </>
   873→              )}
   874→              <div className="grid gap-4 grid-cols-2">
   875→                <div className="space-y-2">
   876→                  <Label>Warranty (months)</Label>
   877→                  <Input type="number" value={form.warrantyMonths as string || ''} onChange={(e) => updateField('warrantyMonths', e.target.value)} placeholder="e.g. 12" />
   878→                </div>
   879→                <div className="space-y-2">
   880→                  <Label>Warranty Info</Label>
   881→                  <Input value={form.warrantyInfo as string || ''} onChange={(e) => updateField('warrantyInfo', e.target.value)} />
   882→                </div>
   883→              </div>
   884→            </CardContent>
   885→          </Card>
   886→
   887→          {/* SEO */}
   888→          <Card>
   889→            <CardHeader><CardTitle className="text-base">SEO</CardTitle></CardHeader>
   890→            <CardContent className="space-y-4">
   891→              <div className="space-y-2">
   892→                <Label>SEO Title</Label>
   893→                <Input value={(form.seoTitle as string) || ''} onChange={(e) => updateField('seoTitle', e.target.value)} />
   894→              </div>
   895→              <div className="space-y-2">
   896→                <Label>SEO Description</Label>
   897→                <Textarea value={(form.seoDescription as string) || ''} onChange={(e) => updateField('seoDescription', e.target.value)} rows={3} />
   898→              </div>
   899→              <div className="space-y-2">
   900→                <Label>Meta Keywords</Label>
   901→                <Input value={(form.metaKeywords as string) || ''} onChange={(e) => updateField('metaKeywords', e.target.value)} />
   902→              </div>
   903→            </CardContent>
   904→          </Card>
   905→
   906→          {/* Metadata */}
   907→          <Card>
   908→            <CardHeader><CardTitle className="text-base">Metadata</CardTitle></CardHeader>
   909→            <CardContent className="space-y-2 text-sm text-muted-foreground">
   910→              <p><span className="font-medium text-foreground">ID:</span> {product.id}</p>
   911→              <p><span className="font-medium text-foreground">Created:</span> {new Date(product.createdAt).toLocaleString()}</p>
   912→              <p><span className="font-medium text-foreground">Updated:</span> {new Date(product.updatedAt).toLocaleString()}</p>
   913→              {product.publishedAt && <p><span className="font-medium text-foreground">Published:</span> {new Date(product.publishedAt).toLocaleString()}</p>}
   914→              <p><span className="font-medium text-foreground">Currency:</span> {product.currency}</p>
   915→              <p><span className="font-medium text-foreground">Type:</span> {product.productType}</p>
   916→            </CardContent>
   917→          </Card>
   918→        </div>
   919→      </div>
   920→
   921→      {/* Add Image Dialog */}
   922→      <Dialog open={imageDialogOpen} onOpenChange={setImageDialogOpen}>
   923→        <DialogContent>
   924→          <DialogHeader>
   925→            <DialogTitle>Add Image</DialogTitle>
   926→            <DialogDescription>Enter the image URL and metadata. The image will be added with PENDING status.</DialogDescription>
   927→          </DialogHeader>
   928→          <div className="space-y-4">
   929→            <div className="space-y-2">
   930→              <Label>Image URL *</Label>
   931→              <Input value={newImageUrl} onChange={(e) => setNewImageUrl(e.target.value)} placeholder="https://..." />
   932→            </div>
   933→            <div className="space-y-2">
   934→              <Label>Alt Text</Label>
   935→              <Input value={newImageAlt} onChange={(e) => setNewImageAlt(e.target.value)} placeholder="Description for accessibility" />

   937→            <div className="space-y-2">

   941→                <SelectContent>
   942→                  {['FRONT', 'BACK', 'SIDE', 'PORTS', 'PACKAGING', 'ACCESSORIES', 'CLOSEUP', 'LIFESTYLE', 'KEYBOARD', 'DISPLAY', 'BOTTOM', 'CHARGER', 'SCREEN', 'OTHER'].map((t) => (
   943→                    <SelectItem key={t} value={t}>{t.charAt(0) + t.slice(1).toLowerCase()}</SelectItem>
   944→                  ))}
   945→                </SelectContent>
   946→              </Select>
   947→            </div>
   948→          </div>
   949→          <DialogFooter>
   950→            <Button variant="outline" onClick={() => setImageDialogOpen(false)}>Cancel</Button>
   951→            <Button onClick={handleAddImage} disabled={!newImageUrl.trim()}>Add Image</Button>
   952→          </DialogFooter>
   953→        </DialogContent>
   954→      </Dialog>
   955→    </div>
   956→  )
