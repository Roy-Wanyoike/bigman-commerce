   200→   200→function flattenCategories(nodes: FlatCategory[], prefix = ''): { id: string; name: string }[] {
   201→   201→  const result: { id: string; name: string }[] = []
   202→   202→  for (const node of nodes) {
   203→   203→    const label = prefix ? `${prefix} > ${node.name}` : node.name
   204→   204→    result.push({ id: node.id, name: label })
   205→   205→    if (node.children?.length) {
   206→   206→      result.push(...flattenCategories(node.children, label))
   207→   207→    }
   208→   208→  }
   209→   209→  return result
   210→   210→}

   212→   212→function getStockInfo(stockCount: number, lowStockThreshold: number, trackInventory: boolean) {
   213→   213→  if (!trackInventory) return { label: 'Not Tracked', color: 'text-muted-foreground', bg: 'bg-gray-50' }
   214→   214→  if (stockCount <= 0) return { label: 'Out of Stock', color: 'text-red-700', bg: 'bg-red-50' }
   215→   215→  if (stockCount <= lowStockThreshold) return { label: 'Low Stock', color: 'text-amber-700', bg: 'bg-amber-50' }
   216→   216→  return { label: 'In Stock', color: 'text-emerald-700', bg: 'bg-emerald-50' }
   217→   217→}
